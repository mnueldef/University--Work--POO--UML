' CLASES DE PERSONAS


' =========================


abstract class Persona {


    - documento : String


    - nombre : String


    + Persona(documento, nombre)


    + getDocumento() : String


    + getNombre() : String


    + setNombre(nombre) : void


    + getDiasMaximosPrestamo() : int


    + getMaximoEquipos() : int


}


class Estudiante {


    - programaAcademico : String


    + Estudiante(documento, nombre, programaAcademico)


    + getProgramaAcademico() : String


    + getDiasMaximosPrestamo() : int


    + getMaximoEquipos() : int


}


class Docente {


    - dependencia : String


    + Docente(documento, nombre, dependencia)


    + getDependencia() : String


    + getDiasMaximosPrestamo() : int


    + getMaximoEquipos() : int


}


Persona <|-- Estudiante


Persona <|-- Docente

' CLASES DE EQUIPOS


' =========================


abstract class Equipo {


    - codigo : String


    - marca : String


    - disponible : boolean


    + Equipo(codigo, marca)


    + getCodigo() : String


    + getMarca() : String


    + isDisponible() : boolean


    + cambiarDisponibilidad(disponible) : void


    + getMultaPorDia() : double


}


class Portatil {


    - memoriaRAM : int


    - cargador : Cargador


    + Portatil(codigo, marca, memoriaRAM, serial, vatios)


    + getMemoriaRAM() : int


    + getCargador() : Cargador


    + getMultaPorDia() : double


}


class Proyector {


    - lumenes : int


    + Proyector(codigo, marca, lumenes)


    + getLumenes() : int


    + getMultaPorDia() : double


}


class Tablet {


    + Tablet(codigo, marca)


    + getMultaPorDia() : double


}


class Cargador {


    - serial : String


    - vatios : int


    + Cargador(serial, vatios)


    + getSerial() : String


    + getVatios() : int


}


Equipo <|-- Portatil


Equipo <|-- Proyector


Equipo <|-- Tablet


Portatil *-- Cargador


' =========================


' CLASE PRESTAMO


' =========================


class Prestamo {


    - contador : int


    - numero : int


    - persona : Persona


    - equipos : ArrayList<Equipo>


    - diasUsados : int


    - diasRetraso : int


    - devuelto : boolean


    + Prestamo(persona, equipos)


    + getNumero() : int


    + getPersona() : Persona


    + getEquipos() : ArrayList<Equipo>


    + devolver(diasUsados) : void


    + calcularMulta() : double


    + isDevuelto() : boolean


}


Prestamo o-- Persona


Prestamo o-- "1..*" Equipo
